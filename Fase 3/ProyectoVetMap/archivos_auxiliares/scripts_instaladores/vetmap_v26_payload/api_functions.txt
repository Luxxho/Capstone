// VETMAP V26 - Repartidores en Tienda (requieren token de tienda).
export async function getStoreDeliveryDrivers(token) {
  const r = await fetch(`${API_URL}/api/store/delivery/drivers`, { headers: { Authorization: `Bearer ${token}` } });
  if (!r.ok) throw new Error(await storeApiError(r, "No se pudieron cargar los repartidores"));
  return r.json();
}
export async function createStoreDeliveryDriver(token, data) {
  const r = await fetch(`${API_URL}/api/store/delivery/drivers`, {
    method: "POST", headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" }, body: JSON.stringify(data),
  });
  if (!r.ok) throw new Error(await storeApiError(r, "No se pudo registrar el repartidor"));
  return r.json();
}
export async function setStoreDeliveryDriverActive(token, id, active) {
  const r = await fetch(`${API_URL}/api/store/delivery/drivers/${id}/active`, {
    method: "PATCH", headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" }, body: JSON.stringify({ active }),
  });
  if (!r.ok) throw new Error(await storeApiError(r, "No se pudo actualizar el repartidor"));
  return r.json();
}
